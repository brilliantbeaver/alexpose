% Preamble for the PDF build of the deck only. The HTML and PowerPoint builds ignore it,
% which keeps output-specific typesetting out of the shared Markdown source.
\usepackage{etoolbox}

% The appendix shell transcripts are wider than a 16:9 frame at body size, so they are
% set small enough to fit rather than allowed to run past the right margin.
\AtBeginEnvironment{Shaded}{\tiny}
\AtBeginEnvironment{verbatim}{\tiny}

% Long reference lines should stretch rather than overflow.
\setlength{\emergencystretch}{3em}

% Artifact names and the 64-character experiment fingerprint are single unbreakable
% tokens that are wider than the column holding them. Allowing a break after any
% character lets them wrap instead of running off the edge of the frame.
\usepackage{seqsplit}
\let\originaltexttt\texttt
\renewcommand{\texttt}[1]{\originaltexttt{\seqsplit{#1}}}

% Figure captions are supporting text, so they sit below the body size.
\setbeamerfont{caption}{size=\scriptsize}

% Tables carry several appendix slides, and at body size they crowd the frame.
\AtBeginEnvironment{longtable}{\scriptsize}
\AtBeginEnvironment{tabular}{\scriptsize}
